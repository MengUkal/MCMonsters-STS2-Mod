# 怪物遭遇代码总览（MCMonsters，快照含 v1.7.0 三层改动）

> 目录：`src\MCMonsters\Encounters\`（30个遭遇类） + `src\MCMonsters\Patches\ActEncounterPatches.cs`（池注入）
> 注：本文档生成于 1.5.0，已把 v1.7.0 的三层改动（末影人单怪进池 / 潜影贝起手错开）同步进来；
> 其余章节以源码为准。

## 一、遭遇池注入补丁（核心）

文件：`src\MCMonsters\Patches\ActEncounterPatches.cs`

```csharp
using System.Collections.Generic;
using MegaCrit.Sts2.Core.Models;
using HarmonyLib;

namespace MCMonsters.Patches;

/// <summary>
/// 把MC遭遇注入各幕遭遇池。
/// Act1弱怪池（IsWeak=true）：僵尸 → 骷髅 → 双蜘蛛
/// Act1强怪池（IsWeak=false）：僵尸+骷髅+蜘蛛 → 苦力怕+骷髅
/// Act2池：双烈焰人 / 双僵尸猪人 / 双凋零骷髅 / 猪灵蛮兵
/// Act3池：单末影人(v1.7.0) / 双潜影贝 / 双女巫 / 双掠夺者
/// 注：末影人自 v1.7.0 起以单怪进池；其余单怪版 *Solo 均不进池（仅控制台测试用）。
/// </summary>
[HarmonyPatch]
public static class ActEncounterPatches
{
    [HarmonyPostfix]
    [HarmonyPatch(typeof(ActModel), "get_AllEncounters")]
    private static void InjectMcEncounters(ActModel __instance, ref IEnumerable<EncounterModel> __result)
    {
        if (__instance.Index == 0)
        {
            __result = __result.Concat(new EncounterModel[]
            {
                // 弱怪池（IsWeak=true，顺序即用户指定的从左到右）
                ModelDb.Encounter<Encounters.ZombieSolo>(),
                ModelDb.Encounter<Encounters.SkeletonSolo>(),
                ModelDb.Encounter<Encounters.SpiderPair>(),
                // 强怪池（IsWeak=false）
                ModelDb.Encounter<Encounters.ZombieSkeletonSpider>(),
                ModelDb.Encounter<Encounters.CreeperSkeleton>(),
                // 精英与BOSS（之前漏加！）
                ModelDb.Encounter<Encounters.ChickenJockeyElite>(),
                ModelDb.Encounter<Encounters.WardenBoss>()
            });
        }
        else if (__instance.Index == 1)
        {
            __result = __result.Concat(new EncounterModel[]
            {
                ModelDb.Encounter<Encounters.BlazePair>(),
                ModelDb.Encounter<Encounters.ZombiePiglinPair>(),
                ModelDb.Encounter<Encounters.WitherSkeletonPair>(),
                ModelDb.Encounter<Encounters.PiglinBruteSolo>(),
                ModelDb.Encounter<Encounters.PiglinBrutePair>(),
                // 组合遭遇
                ModelDb.Encounter<Encounters.BlazeWitherSkeleton>(),
                ModelDb.Encounter<Encounters.WitherSkeletonTriple>(),
                ModelDb.Encounter<Encounters.BlazeZombiePiglin>(),
                ModelDb.Encounter<Encounters.ZombiePiglinWitherSkeleton>(),
                // 精英与BOSS（之前漏加！）
                ModelDb.Encounter<Encounters.GhastElite>(),
                ModelDb.Encounter<Encounters.WitherBossEncounter>()
            });
        }
        else if (__instance.Index == 2)
        {
            __result = __result.Concat(new EncounterModel[]
            {
                ModelDb.Encounter<Encounters.EndermanPair>(),
                ModelDb.Encounter<Encounters.ShulkerPair>(),
                ModelDb.Encounter<Encounters.WitchPair>(),
                ModelDb.Encounter<Encounters.PillagerPair>(),
                // 精英与BOSS
                ModelDb.Encounter<Encounters.GiantElite>(),
                ModelDb.Encounter<Encounters.PillagerRaidElite>(),
                ModelDb.Encounter<Encounters.EnderDragonBoss>()
            });
        }
    }
}
```

## 二、遭遇类示例（完整模板）

文件：`src\MCMonsters\Encounters\ZombieSolo.cs`（最简单怪模板）

```csharp
using System.Collections.Generic;
using MegaCrit.Sts2.Core.Models;
using MegaCrit.Sts2.Core.Models.Monsters;
using MegaCrit.Sts2.Core.Rooms;
using MCMonsters.Monsters;

namespace MCMonsters.Encounters;

/// <summary>
/// 遭遇：一只僵尸（普通怪，非weak池）。
/// 类名即 ModelId（slug: zombie_solo），无需手动注册——
/// ModelDb.Init() 会扫描MOD程序集中所有 EncounterModel 子类自动注册。
/// </summary>
public sealed class ZombieSolo : EncounterModel
{
    private static readonly MonsterModel[] _zombies =
    {
        ModelDb.Monster<Zombie>()
    };

    public override RoomType RoomType => RoomType.Monster;

    public override bool IsWeak => true;

    public override IEnumerable<MonsterModel> AllPossibleMonsters => _zombies;

    protected override IReadOnlyList<(MonsterModel, string?)> GenerateMonsters()
    {
        return new (MonsterModel, string?)[]
        {
            (ModelDb.Monster<Zombie>().ToMutable(), null)
        };
    }
}
```

### ChickenJockeyElite（⭐精英 · 🔴强怪池）

```csharp
using System.Collections.Generic;
using MegaCrit.Sts2.Core.Models;
using MegaCrit.Sts2.Core.Models.Monsters;
using MegaCrit.Sts2.Core.Rooms;
using MCMonsters.Monsters;

namespace MCMonsters.Encounters;

/// <summary>
/// 遭遇：小僵尸鸡骑士（第1幕精英，双命）。
/// RoomType.Elite → ActModel.AllEliteEncounters 自动从 AllEncounters 过滤进精英池。
/// Slots 提供召唤槽位：鸡骑士本体占 jockey 槽，开场召唤的僵尸/骷髅占 minion1/2 槽
/// （不提供空槽位时 EncounterModel.GetNextSlot 拿不到槽，召唤会静默失败）。
/// </summary>
public sealed class ChickenJockeyElite : EncounterModel
{
    private const string JockeySlot = "jockey";
    private const string MinionSlot1 = "minion1";
    private const string MinionSlot2 = "minion2";

    private static readonly MonsterModel[] _monsters =
    {
        ModelDb.Monster<ChickenJockey>()
    };

    public override RoomType RoomType => RoomType.Elite;

    public override bool HasScene => true;

    public override IReadOnlyList<string> Slots => new[] { JockeySlot, MinionSlot1, MinionSlot2 };

    public override IEnumerable<MonsterModel> AllPossibleMonsters => _monsters;

    protected override IReadOnlyList<(MonsterModel, string?)> GenerateMonsters()
    {
        return new (MonsterModel, string?)[]
        {
            (ModelDb.Monster<ChickenJockey>().ToMutable(), JockeySlot)
        };
    }
}
```

```csharp
using System.Collections.Generic;
using MegaCrit.Sts2.Core.Models;
using MegaCrit.Sts2.Core.Models.Monsters;
using MegaCrit.Sts2.Core.Rooms;
using MCMonsters.Monsters;

namespace MCMonsters.Encounters;

/// <summary>
/// 遭遇：双潜影贝（第3幕池）。MC末地城的潜影贝成排附着——
/// 双贝壳起手错开：一只以闭合防御开始，一只以开壳弹幕开始，
/// 之后各自闭合/开壳交替，消耗战拉满。
/// 类名即 ModelId（slug: shulker_pair）。
/// </summary>
public sealed class ShulkerPair : EncounterModel
{
    public override RoomType RoomType => RoomType.Monster;

    public override IEnumerable<MonsterModel> AllPossibleMonsters => new MonsterModel[]
    {
        ModelDb.Monster<Shulker>(),
        ModelDb.Monster<Shulker>()
    };

    protected override IReadOnlyList<(MonsterModel, string?)> GenerateMonsters()
    {
        var first = (Shulker)ModelDb.Monster<Shulker>().ToMutable();
        var second = (Shulker)ModelDb.Monster<Shulker>().ToMutable();
        // 两只初始意图必为一格挡一攻击：随机谁先开壳（遭遇级RNG，同种子可复现）
        bool firstBarrage = Rng.NextBool();
        first.StartOnBarrage = firstBarrage;
        second.StartOnBarrage = !firstBarrage;
        return new (MonsterModel, string?)[]
        {
            (first, (string?)null),
            (second, (string?)null)
        };
    }
}
```
